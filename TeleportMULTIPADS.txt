#==================================================
# HOLO MANAGER
#==================================================

function number nextHolo() {
    HoloIdx++
    return HoloIdx
}

function vector lerpVector(M:number,G:vector,C:vector){
    return G + M * (C-G)
}

function entity holoMake(
    Parent:entity,
    Offset:vector,
    Scale:vector,
    Rotation:angle,
    Color:vector,
    Alpha:number,
    Shape:string,
    Mat:string
) {
    local Idx = nextHolo()
    local Holo = holoCreate(Idx, Parent:toWorld(Offset), Scale, Parent:toWorld(Rotation), Color, Shape)
    holoAlpha(Idx, Alpha)
    if(Mat != "") { holoMaterial(Idx, Mat) }
    Holo:parentTo(Parent)
    return Holo
}

#==================================================
# COOLDOWN
#==================================================

function number isOnCooldown(Ply:entity) {
    local SteamID = Ply:steamID()
    if(!COOLDOWNS:exists(SteamID)) { return 0 }
    return (curtime() - COOLDOWNS[SteamID, number]) < COOLDOWN_TIME
}

function number getCooldownRemaining(Ply:entity) {
    local SteamID = Ply:steamID()
    if(!COOLDOWNS:exists(SteamID)) { return 0 }
    return max(0, COOLDOWN_TIME - (curtime() - COOLDOWNS[SteamID, number]))
}

function void setCooldown(Ply:entity) {
    local SteamID = Ply:steamID()
    COOLDOWNS[SteamID, number] = curtime()
    if(SelectedPad:isValid()) {
        dsSend("tp_cooldown_sync", PAD_NAME, table("steamid" = SteamID, "partner" = SelectedPad))
    }
}

#==================================================
# PARTICLES
#==================================================

function void burstParticles(Origin:vector) {
    if(!EnableParticles) { return }
    particleSetMaterial("sprites/glow04_noz")
    particleSetColor(ParticleColor)
    particleSetPitch(0)
    particleSetStartAlpha(255)
    particleSetEndAlpha(0)
    particleSetStartSize(ParticleScale)
    particleSetEndSize(ParticleScale_END)
    particleSetDuration(ParticleDuration)
    particleBounce(0)
    particleGravity(vec())
    particleCollision(0)
    local GoldenRatio = (1 + sqrt(5)) / 2
    for(I=1, ParticleCount) {
        local Theta = 360 * I / GoldenRatio
        local Phi = acos(1 - 2*I / ParticleCount)
        local Dir = vec(sin(Phi)*cos(Theta), sin(Phi)*sin(Theta), cos(Phi))
        particleSetPosition(Origin + Dir * 30)
        particleSetVelocity(Dir * Vel_spritres)
        particleCreate()
    }
}

#==================================================
# PAD / WHITELIST
#==================================================

function number isOnPad(Ply:entity) {
    return Ply:pos():distance(entity():pos()) <= UseTriggerDist
}

function number isWhitelisted(Ply:entity) {  
    if(!WLMode) { return 1 }
    if(!Ply:isValid()) { return 0 }
    foreach(_:number, Allowed:entity = Whitelist) {         
        if(!Allowed:isValid()) { continue }
        if(Allowed == Ply) { return 1 }
    }
    return 0
}

function void doTeleportSound() {
    if(PendingTpSnd == "") { return }
    entity():emitSoundStop(SndChargeup)
    entity():emitSound(PendingTpSnd, SndTeleportVol, SndTeleportPitch, 1)
    PendingTpSnd = ""
}

function void queueTeleportSound(Snd:string) {
    if(Snd == "") { return }
    PendingTpSnd = Snd
    stoptimer("tp_sound")
    timer("tp_sound", SndTeleportDelay, function() { doTeleportSound() })
}

function void emitPadSound(Snd:string, Vol:number, Pitch:number) {
    entity():emitSound(Snd, Vol, Pitch, 1)
}

function number tubeCenterZ() {
    return TubeBaseZ + Height/2
}

function vector tubeScale() {
    return vec(Holo_tube_w, Holo_tube_w, Height) / 190.3
}

# HOLO BUTTON
#==================================================

function number isLookingAt(HoloIdx:number, Radius:number, Ply:entity) {
    if(!Ply:isValid()) { return 0 }
    if(!holoEntity(HoloIdx):isValid()) { return 0 }
    local HoloPos = holoEntity(HoloIdx):pos()
    local EyePos  = Ply:shootPos()
    local EyeDir  = Ply:eye()
    local Dist    = EyePos:distance(HoloPos)
    local AimPt   = EyePos + EyeDir * Dist
    return AimPt:distance(HoloPos) < Radius
}

function void setButtonColor(State:number) {
    if(!holoEntity(50):isValid()) { return }
    if(State == 1) {
        holoEntity(50):setSubMaterial(2, "bobsters_trains/br_apple_green")
    } elseif(State == 2) {
        holoEntity(50):setSubMaterial(2, "bobsters_trains/br_buffer_red")
    } else {
        holoEntity(50):setSubMaterial(2, "")
    }
}

function void doButtonReset() {
    setButtonColor(0)
}

    foreach(_:string, _:string = Partners) { return 1 }
    return 0
}

    foreach(IDStr:string, _:string = Partners) {
        return entity(IDStr:toNumber())
    }
    return noentity()
}

function number partnerCount() {
    local N = 0
    foreach(_:string, _:string = Partners) { N++ }
    return N
}

function void egpMakeButton([Center Size]:vector2, Label:string, PadEnt:entity) {
    EGPIndex++
    local Button = EGP:egpBox(EGPIndex, Center, Size)
    EGP:egpColor(EGPIndex, VguiColorReady)
    EGPIndex++
    local Outline = EGP:egpBoxOutline(EGPIndex, vec2(), Size)
    EGP:egpSize(EGPIndex, 3)
    EGP:egpColor(EGPIndex, EGP_OutlineCol)
    Outline:parentTo(Button)
    EGPIndex++
    local Text = EGP:egpTextLayout(EGPIndex, Label, vec2(), Size)
    EGP:egpSize(EGPIndex, EGP_TextSize * EGP_TextScale)
    EGP:egpAlign(EGPIndex, 1, 1)
    EGP:egpColor(EGPIndex, vec())
    Text:parentTo(Button)
    EGPButtons:pushTable(table("egpobject" = Button, "pad" = PadEnt))
}

function void buildEGPScreen() {
    if(!EnableEGPPicker) { return }
    if(!EGP:entity():isValid()) { return }

    EGP:egpClear()
    EGPIndex   = 0
    EGPButtons = table()
    EGPBuilt   = 1

    EGP:egpBox(100001, vec2(256, 256), vec2(512, 512))
    if(EGP_BgMaterial != "") { EGP:egpMaterial(100001, EGP_BgMaterial) }
    EGP:egpAlign(100001, 1)
    EGP:egpColor(100001, EGP_BgCol)

    local Pads   = array()
    local Labels = array()
    foreach(IDStr:string, Label:string = Partners) {
        local PadEnt = entity(IDStr:toNumber())
        if(!PadEnt:isValid()) { continue }
        Pads:pushEntity(PadEnt)
        Labels:pushString(Label)
    }

    local Count = Pads:count()
    if(!Count) {
        # A textLayout's position is the CENTER of its layout box (textlayout.lua
        # draws at x - w/2, y - h/2), so the box must sit at 256,256 to fill the
        # screen -- vec2(0,0) puts three quarters of it off-screen.
        EGP:egpTextLayout(100002, EGP_UnpairedText, vec2(256, 256), vec2(512, 512))
        EGP:egpSize(100002, EGP_UnpairedTextSize * EGP_UnpairedTextScale)
        EGP:egpAlign(100002, 1, 1)
        EGP:egpColor(100002, EGP_UnpairedTextCol)
        return
    }

    local Padding    = 20
    local NumCols    = ceil(Count / max(1, EGP_ButtonsPerRow))
    local ActualRows = ceil(Count / NumCols)

    local BtnW = ((512 - Padding*2) - (NumCols - 1) * EGP_ColSpacing) / NumCols * EGP_ButtonScaleX
    local BtnH = ((512 - Padding*2) - (ActualRows - 1) * EGP_ButtonSpacing) / ActualRows * EGP_ButtonScaleY

    local BlockW = NumCols * BtnW + (NumCols - 1) * EGP_ColSpacing
    local BlockH = ActualRows * BtnH + (ActualRows - 1) * EGP_ButtonSpacing
    local StartX = (512 - BlockW) / 2 + BtnW / 2
    local StartY = (512 - BlockH) / 2 + BtnH / 2

    for(I = 1, Count) {
        local Col = 0
        local Row = 0
        if(EGP_FillLeftToRight) {
            Col = (I - 1) % NumCols
            Row = floor((I - 1) / NumCols)
        } else {
            Col = floor((I - 1) / ActualRows)
            Row = (I - 1) % ActualRows
        }
        egpMakeButton(
            vec2(StartX + Col * (BtnW + EGP_ColSpacing), StartY + Row * (BtnH + EGP_ButtonSpacing)),
            vec2(BtnW, BtnH),
            Labels[I, string],
            Pads[I, entity]
        )
    }
}

function void updateEGPColors() {
    if(!EnableEGPPicker) { return }
    if(!EGPBuilt) { return }
    if(!EGP:entity():isValid()) { return }

    local Col = VguiColorReady
    if(Recharging == 1) { Col = VguiColorRecharge }
    foreach(_:number, Btn:table = EGPButtons) {
        local Xeo = Btn["egpobject", egpobject]
        if(EGP:egpHasObject(Xeo) != 1) { continue }
        Xeo:setColor(Col)
    }
}

function void scheduleEGPRebuild() {
    if(!EnableEGPPicker) { return }
    stoptimer("egp_rebuild")
    timer("egp_rebuild", EGP_RebuildDelay, function() { buildEGPScreen() })
}

# Waits for the spawned screen prop to go valid, parents/statics it to the pad,
# then draws the buttons. Re-arms itself instead of assuming the prop exists on
# the tick spawnEgp() ran.
function void egpSetup() {
    if(!EnableEGPPicker) { return }
    if(!EGP:entity():isValid()) {
        stoptimer("egp_setup")
        timer("egp_setup", EGP_SetupDelay, function() { egpSetup() })
        return
    }
    if(!EGP:entity():parent():isValid()) {
        EGP:entity():parentTo(entity())
        EGP:entity():propStatic(1)
        EGP:entity():noCollideAll(1)
    }
    buildEGPScreen()
}

    if(PickerOpen) { closePickerVGUI() }
    if(!Ply:isValid()) { return }
    if(timerExists("tp_charge")) { return }
    if(partnerCount() <= 1) { return }
    
    PickerOpen = 1
    PickerPlayer = Ply
    setButtonColor(1)

    PickerFrame = dframe(200)
    PickerFrame:setTitle("  SELECT DESTINATION")
    local Count = partnerCount()
    PickerFrame:setSize(280, 45 + Count * 80)
    PickerFrame:center()
    PickerFrame:makePopup()
    PickerFrame:showCloseButton(1)
    PickerFrame:setColor(20, 20, 20, 190)
    PickerFrame:setPlayers(array(Ply))
    PickerFrame:create()

    PickerButtons = table()
    PickerButtonRefs = table()
    local BtnIdx = 0
    foreach(IDStr:string, Label:string = Partners) {
        local PadEnt = entity(IDStr:toNumber())
        if(!PadEnt:isValid()) { continue }

function void doVisualUpdate() {
    Ops = ops()
    local Pulse = (sin(curtime()*LerpSpeed)+1)/2
    local IsCharging = timerExists("tp_charge")
    local HasPartner = hasAnyPartner()
    local RingCol = vec(0)

    if(Ring1Idx > 0) {
        if(IsCharging) {
            RingCol = vec(255,185,0)
            holoColor(Ring1Idx, RingCol)
        } elseif(!HasPartner) {
            RingCol = lerp(vec(120,0,0), vec(255,0,0), Pulse)
            holoColor(Ring1Idx, RingCol)
        } elseif(!RingRGBMode) {
            RingCol = lerp(Conected_col, Conected_col, Pulse)
            holoColor(Ring1Idx, RingCol)
        } else {
            if(!RGBTimerRunning) { startRGBTimer() }
        }
    }

    if(H:exists("Rift1") && H["Rift1",entity]:isValid()) {
        H["Rift1",entity]:setAlpha(HasPartner ? 255 : 0)
    }
    if(H:exists("Rift2") && H["Rift2",entity]:isValid()) {
        H["Rift2",entity]:setAlpha(HasPartner ? 255 : 0)
    }

    if(StepHoloPulse != 0) {
        local PulseCol = StepHoloPulse == 1 ? StepHoloCol2 : StepHoloDenied
        holoColor(35, PulseCol)
    }
    updatePickerButtonColors()
    updateEGPColors()
}

#==================================================
# NPC / PROP SCAN (infinite)
#==================================================

function void doNpcPropScan() {
    if(!ChargeActive) {
        if(EnablePropTeleport) {
            findIncludeClass("prop_physics")
            findInCone(entity():pos(), vec(0,0,1), UseTriggerDist, ConeDegrees)
            foreach(_:number, Prop:entity = findToArray()) {
                if(!Prop:isValid()) { continue }
                if(Prop == entity()) { continue }
                local PropID = Prop:id() + ""
                if(!PropsOnPad:exists(PropID)) {
                    local RecentlyTeleported = PropCooldowns:exists(PropID) && (curtime() - PropCooldowns[PropID, number]) < COOLDOWN_TIME
                    PropsOnPad[PropID, number] = curtime()
                    StepHoloPulse = 1
                    if(!RecentlyTeleported) {
                        emitPadSound(SndWelcome, SndWelcomeVol, SndWelcomePitch)
                    }
                }
            }
            local PropsToRemove = array()
            foreach(PropID:string, _:number = PropsOnPad) {
                local Prop = entity(PropID:toNumber())
                if(!Prop:isValid() | Prop:pos():distance(entity():pos()) > LeavePadDist) {
                    if((curtime() - PropsOnPad[PropID, number]) > 0.5) {
                        PropsToRemove:pushString(PropID)
                    }
                }
            }
            foreach(_:number, PropID:string = PropsToRemove) { PropsOnPad:remove(PropID) }
            if(!NPCsOnPad:count() & !PropsOnPad:count() & !PlayersOnPad:count()) {
                StepHoloPulse = 0
                holoColor(35, StepHoloCol1)
            }
        }

        if(EnableNPCTeleport) {
            findIncludeClass("npc")
            findInCone(entity():pos(), vec(0,0,1), UseTriggerDist, ConeDegrees)
            foreach(_:number, Npc:entity = findToArray()) {
                if(!Npc:isValid()) { continue }
                local NpcID = Npc:id() + ""
                if(!NPCsOnPad:exists(NpcID)) {
                    NPCsOnPad[NpcID, number] = 1
                    StepHoloPulse = 1
                    emitPadSound(SndWelcome, SndWelcomeVol, SndWelcomePitch)
                }
            }
            local NpcsToRemove = array()
            foreach(NpcID:string, _:number = NPCsOnPad) {
                local Npc = entity(NpcID:toNumber())
                if(!Npc:isValid() | Npc:pos():distance(entity():pos()) > LeavePadDist) {
                    NpcsToRemove:pushString(NpcID)
                }
            }
            foreach(_:number, NpcID:string = NpcsToRemove) { NPCsOnPad:remove(NpcID) }
            if(!NPCsOnPad:count() & !PropsOnPad:count() & !PlayersOnPad:count()) {
                StepHoloPulse = 0
                holoColor(35, StepHoloCol1)
            }
        }
    }
}

#==================================================
# PAD GREET
#==================================================

function void doPadGreet() {
    if(!Tele_Players) { exit() }
    foreach(SteamID:string, Ply:entity = PendingGreets) {
        if(!Ply:isValid()) { PendingGreets:remove(SteamID) continue }
        if(TeleportedPlayers:exists(SteamID)) { PendingGreets:remove(SteamID) continue }
        if(Ply:pos():distance(entity():pos()) > UseTriggerDist) { PendingGreets:remove(SteamID) continue }

        if(!isWhitelisted(Ply)) {
            chatPrint(Ply, vec(255,0,0), "WARNING! ", vec(255,100,100), Ply:name(), vec(255,0,0), "  NOT ALLOWED!")
            emitPadSound(SndDenied, SndDeniedVol, SndDeniedPitch)
            if(Deny_Type==1){ Ply:plySetHealth(1) boomCustom("cball_explode",Ply:boxCenterW(),10,2) }
            if(Deny_Type==2){ Ply:plySpawn() }
            if(Deny_Type==3){ Ply:kill() }
            if(Deny_Type==4){ Ply:killSilent() }
            StepHoloPulse = 2
        } else {
            local UseHint = ""
            if(EnableHoloButton) { UseHint = "hit E while looking at the Holo Button to use" }
            elseif(EnableUseTrigger) { UseHint = "look at the pad ring and hit E to use" }
            else { UseHint = "awaiting wire trigger to use" }
            chatPrint(Ply, vec(0,255,100), "Hello! ", vec(255,255,255), Ply:name(), vec(0,200,255), " " + UseHint)
            emitPadSound(SndWelcome, SndWelcomeVol, SndWelcomePitch)
            StepHoloPulse = 1
        }
        PendingGreets:remove(SteamID)
    }
}

#==================================================
# PAD PRESENCE (clk timer + tick backup like teli WIP.txt)
#==================================================

function void doPadPresence() {
    if(Tele_Players) {
        foreach(_:number, Ply:entity = players()) {
            if(!Ply:isValid()) { continue }
            local SteamID = Ply:steamID()
            local Dist = Ply:pos():distance(entity():pos())

            if(!PlayersOnPad:exists(SteamID) & Dist <= UseTriggerDist) {
                PlayersOnPad[SteamID, number] = curtime()
                if(!TeleportedPlayers:exists(SteamID)) {
                    PendingGreets[SteamID, entity] = Ply
                    if(!timerExists("pad_greet")) { timer("pad_greet", PadGreetDelay, function() { doPadGreet() }) }
                    StepHoloPulse = 1
                }
            } elseif(PlayersOnPad:exists(SteamID) & Dist > LeavePadDist) {
                local ArrivalTime = PlayersOnPad[SteamID, number]
                if((curtime() - ArrivalTime) < 1) { continue }
                PlayersOnPad:remove(SteamID)
                TeleportedPlayers:remove(SteamID)
                PendingGreets:remove(SteamID)
                if(PickerPlayer == Ply) { closePickerVGUI() }
                PrevKeyUse[SteamID, number] = 0
                if(PlayersOnPad:count() == 0) {
                    StepHoloPulse = 0
                    if(holoEntity(35):isValid()) { holoEntity(35):setColor(StepHoloCol1) }
                }
            }
        }
    }

    local GravTargets = ChargeActive ? ChargePlayers : (EnableRiseRing ? HeldPlayers : array())
    foreach(_:number, Ply:entity = GravTargets) {
        if(!Ply:isValid()) { continue }
        local PadCenter = entity():toWorld(vec(0, 0, 10))
        local ToCenter  = PadCenter - Ply:pos()
        local Dist      = ToCenter:length()
        if(Dist < 2) { continue }
        local TargetVel = ToCenter:normalized() * clamp(Dist * GravWellDistMult, GravWellMinSpeed, GravWellSnapSpeed)
        Ply:plyApplyForce(TargetVel - Ply:vel())
    }
}

#==================================================
# SPRITE FADE (self-rearming)
#==================================================

function void doSpriteFade() {
    SpriteAlpha  *= SpriteFadeSpeed
    SpriteAlpha2 *= SpriteFadeSpeed
    LightFade1   *= LightFadeSpeed

    local SpriteFrac = clamp(SpriteAlpha / 255, 0, 1)
    local LightFrac1 = clamp(LightFade1 / 25, 0, 1)
    local SpriteCol = lerpVector(SpriteFrac, vec(0,0,0), Sprite_Ball_Col)
    local LightCol1 = lerpVector(LightFrac1, vec(0,0,0), Light_Col)

    if(EnableSprites) {
        spriteSetColor(1, SpriteCol)
        spriteSetColor(3, SpriteCol)
        spriteSetAlpha(1, SpriteAlpha)
        spriteSetAlpha(3, SpriteAlpha)
    }
    lightColor(1, LightCol1)
    lightBrightness(1, LightFade1)

    if(EnableTubes & TubesCreated) {
        holoAlpha(100, SpriteAlpha)
        holoShadow(100,0)
    }

    if(SpriteAlpha <= 1 & SpriteAlpha2 <= 1 & LightFade1 <= 0.1) {
        SpriteAlpha = 0
        SpriteAlpha2 = 0
        LightFade1 = 0
        if(EnableSprites) {
            spriteEnable(1,0)
            spriteEnable(3,0)
        }
        lightToggle(1,0)
        if(EnableTubes & TubesCreated) {
            holoAlpha(100,0)
            holoShadow(100,0)
        }
    } else {
        stoptimer("sprite_fade")
        timer("sprite_fade", SpriteFadeStepDelay, function() { doSpriteFade() })
    }
}

#==================================================
# COOLDOWN REMIND (self-rearming)
#==================================================

function void doCooldownRemind() {
    local AnyStillOnCooldown = 0
    foreach(_:number, Ply:entity = LastPortedPlayers) {
        if(!Ply:isValid()) { continue }
        if(!isOnCooldown(Ply)) {
            chatPrint(Ply, vec(0,255,255), "[Teleporter] ", vec(0,255,100), "Ready!")
            continue
        }
        local Remaining = ceil(getCooldownRemaining(Ply))
        chatPrint(Ply, vec(0,255,255), "[Teleporter] ", vec(0,200,120), "Cooldown: ", vec(255,255,0), "" + Remaining, vec(0,200,120), " Sec remaining")
        AnyStillOnCooldown = 1
    }
    foreach(_:number, Npc:entity = LastPortedNPCs) {
        if(!Npc:isValid()) { continue }
        local NpcID = Npc:id() + ""
        if(!NpcCooldowns:exists(NpcID)) { continue }
        local Remaining = ceil(max(0, COOLDOWN_TIME - (curtime() - NpcCooldowns[NpcID, number])))
        if(Remaining > 0) {
            printColor(vec(255,200,0), "[Teleporter NPC] ", vec(255,255,255), "", vec(255,255,0), " Cooldown: ", vec(255,255,255), "" + Remaining, vec(255,255,0), " sec")
            AnyStillOnCooldown = 1
        } else {
            printColor(vec(0,255,0), "[Teleporter NPC] ", vec(255,255,255), Npc:name(), vec(0,255,0), " READY!")
        }
    }
    foreach(_:number, Prop:entity = LastPortedProps) {
        if(!Prop:isValid()) { continue }
        local PropID = Prop:id() + ""
        if(!PropCooldowns:exists(PropID)) { continue }
        local Remaining = ceil(max(0, COOLDOWN_TIME - (curtime() - PropCooldowns[PropID, number])))
        if(Remaining > 0) {
            printColor(vec(0,200,255), "[Teleporter Prop] ", vec(255,255,255), "", vec(255,255,0), " Cooldown: ", vec(255,255,255), "" + Remaining, vec(255,255,0), " sec")
            AnyStillOnCooldown = 1
        } else {
            printColor(vec(0,255,0), "[Teleporter Prop] ", vec(255,255,255), Prop:model(), vec(0,255,0), " READY!")
        }
    }
    if(AnyStillOnCooldown) {
        stoptimer("cooldown_remind")
        timer("cooldown_remind", CooldownRemindInterval, function() { doCooldownRemind() })
    }
}

#==================================================
# TP CHARGE (the actual teleport)
#==================================================

function void doTpCharge() {
    if(PickerOpen) { closePickerVGUI() }

    local TpTarget = H["TpTarget",entity]
    local Dest = TpTarget:isValid() ? TpTarget:pos() : SelectedPad:pos() + vec(0,0,10)
    local DestAng = SelectedPad:toWorld(ang(0, PartnerDir, 0))

    foreach(_:number, Ply:entity = ChargePlayers) {
        if(!Ply:isValid()) { continue }
        if(isOnCooldown(Ply)) { continue }
        local SteamID = Ply:steamID()
        dsSend("av_teleported_marker", PAD_NAME, table("steamid" = SteamID, "target" = SelectedPad))
    }

    LastPortedPlayers = array()
    LastPortedNPCs  = array()
    LastPortedProps = array()

    local Snd = Snds[randint(Snds:count()), string]
    local AnyPorted = 0

    foreach(_:number, Ply:entity = ChargePlayers) {
        if(!Ply:isValid()) { continue }
        if(isOnCooldown(Ply)) { continue }
        if(!isWhitelisted(Ply)) { continue }
        setCooldown(Ply)
        Ply:plySetPos(Dest + vec(0, 0, TeleportHeightPlayer))
        if(Forced_direction==1){ Ply:plySetAng(DestAng) }
        local SteamID = Ply:steamID()
        PlayersOnPad:remove(SteamID)
        TeleportedPlayers:remove(SteamID)
        PendingGreets:remove(SteamID)
        LastPortedPlayers:pushEntity(Ply)
        AnyPorted = 1
    }

    foreach(_:number, Npc:entity = ChargeNPCs) {
        if(!Npc:isValid()) { continue }
        local NpcID = Npc:id() + ""
        if(NpcCooldowns:exists(NpcID) && (curtime() - NpcCooldowns[NpcID, number]) < COOLDOWN_TIME) { continue }
        Npc:setPos(Dest + vec(0, 0, TeleportHeightNPC))
        NpcCooldowns[NpcID, number] = curtime()
        dsSend("tp_npc_cooldown_sync", PAD_NAME, table("id" = NpcID, "target" = SelectedPad))
        LastPortedNPCs:pushEntity(Npc)
        AnyPorted = 1
        NPCsOnPad:remove(NpcID)
    }

    foreach(_:number, Prop:entity = ChargeProps) {
        if(!Prop:isValid()) { continue }
        if(Prop == entity()) { continue }
        local PropID = Prop:id() + ""
        if(PropCooldowns:exists(PropID) && (curtime() - PropCooldowns[PropID, number]) < COOLDOWN_TIME) { continue }
        local HalfHeight = (Prop:boxMax():z() - Prop:boxMin():z()) / 2
        Prop:setPos(Dest + vec(0, 0, HalfHeight + TeleportHeightProp))
        PropCooldowns[PropID, number] = curtime()
        dsSend("tp_prop_cooldown_sync", PAD_NAME, table("id" = PropID, "target" = SelectedPad))
        LastPortedProps:pushEntity(Prop)
        AnyPorted = 1
        StepHoloPulse = 0
        PropsOnPad:remove(PropID)
    }

    ChargeNPCs  = array()
    ChargeProps = array()

    if(AnyPorted) {
        queueTeleportSound(Snd)
    }

    StepHoloPulse = 0
    holoColor(35, StepHoloCol1)
    Spooling = 2

    RiseRingDescendReady = 0
    if(RiseRingDescendDelay > 0) {
        stoptimer("ring_descend_delay")
        timer("ring_descend_delay", RiseRingDescendDelay, function() { doRingDescendDelay() })
    } else {
        RiseRingDescendReady = 1
    }

    dsSend("av_spool_sync", PAD_NAME, table("target" = SelectedPad, "state" = 2, "ringZ" = RiseRingCurrentZ))

    stoptimer("cooldown_remind")
    timer("cooldown_remind", CooldownRemindInterval, function() { doCooldownRemind() })

    effect():setOrigin(entity():toWorld(vec(0,0,12)))
    effect():setNormal(vec(0,0,1))

#==================================================
# RUN TELEPORT
#==================================================

function void runTeleport() {
    if(!SelectedPad:isValid()) { exit() }
    if(Recharging == 1) { exit() }
    if(ChargeActive) { exit() }

    local DestBlocked = 0
    BlockedBy = noentity()
    foreach(_:number, OtherPly:entity = players()) {
        if(!OtherPly:isValid()) { continue }
        if(OtherPly:pos():distance(SelectedPad:pos()) <= UseTriggerDist) {
            DestBlocked = 1
            BlockedBy = OtherPly
            break
        }
    }

    if(DestBlocked) {
        entity():emitSound(SndCooldownDeny, SndCooldownDenyVol, SndCooldownDenyPitch)
        local BlockedName = BlockedBy:isValid() ? BlockedBy:name() : "Someone"
        foreach(_:number, Ply2:entity = players()) {
            if(!Ply2:isValid()) { continue }
            if(Ply2:pos():distance(entity():pos()) > UseTriggerDist) { continue }
            if(Ply2 == BlockedBy) { continue }
            chatPrint(Ply2, vec(255,100,0), "[Teleporter] ", vec(255,255,255), "Cannot teleport: ", vec(255,200,0), BlockedName, vec(255,255,255), " is on the destination pad, please step off.")
        }
        if(BlockedBy:isValid() && BlockedBy:pos():distance(entity():pos()) > UseTriggerDist) {
            chatPrint(BlockedBy, vec(255,100,0), "[Teleporter] ", vec(255,255,255), "Please step off the pad, someone is waiting to teleport here!")
        }
        setButtonColor(2)
        stoptimer("button_reset")
        timer("button_reset", ButtonDeniedResetTime, function() { doButtonReset() })
        exit()
    }

    local Found = array()
    if(Tele_Players) {
        local Pool = WLMode ? Whitelist : players()
        foreach(_:number, Ply:entity = Pool) {
            if(!Ply:isValid()) { continue }
            if(Ply:pos():distance(entity():pos()) > UseTriggerDist) { continue }
            if(isOnCooldown(Ply)) {
                emitPadSound(SndCooldownDeny, SndCooldownDenyVol, SndCooldownDenyPitch)
                continue
            }
            Found:pushEntity(Ply)
        }
    }

    ChargeNPCs = array()
    if(EnableNPCTeleport) {
        foreach(NpcID:string, _:number = NPCsOnPad) {
            local Npc = entity(NpcID:toNumber())
            if(!Npc:isValid()) { continue }
            if(Npc:pos():distance(entity():pos()) > UseTriggerDist) { continue }
            if(NpcCooldowns:exists(NpcID) && (curtime() - NpcCooldowns[NpcID, number]) < COOLDOWN_TIME) { continue }
            ChargeNPCs:pushEntity(Npc)
        }
    }

    ChargeProps = array()
    if(EnablePropTeleport) {
        foreach(PropID:string, _:number = PropsOnPad) {
            local Prop = entity(PropID:toNumber())
            if(!Prop:isValid()) { continue }
            if(Prop == entity()) { continue }
            if(Prop:pos():distance(entity():pos()) > UseTriggerDist) { continue }
            if(PropCooldowns:exists(PropID) && (curtime() - PropCooldowns[PropID, number]) < COOLDOWN_TIME) { continue }
            ChargeProps:pushEntity(Prop)
        }
    }

    if(!Found:count() & !ChargeNPCs:count() & !ChargeProps:count()) {
        entity():emitSound(SndCooldownDeny, SndCooldownDenyVol, SndCooldownDenyPitch)
        setButtonColor(2)
        stoptimer("button_reset")
        timer("button_reset", ButtonDeniedResetTime, function() { doButtonReset() })
        exit()
    }

    ChargePlayers = Tele_Players ? Found : array()
    ChargeActive = 1
    stoptimer("npc_prop_scan")

    entity():emitSound(SndChargeup, SndChargeupVol, SndChargeupPitch)
    SelectedPad:emitSound(SndChargeup, SndChargeupVol, SndChargeupPitch)
    setButtonColor(0)

    Spooling = 1

    RiseRingAscendReady = 0
    if(RiseRingAscendDelay > 0) {
        stoptimer("ring_ascend_delay")
        timer("ring_ascend_delay", RiseRingAscendDelay, function() { doRingAscendDelay() })
    } else {
        RiseRingAscendReady = 1
    }

    dsSend("av_spool_sync", PAD_NAME, table("target" = SelectedPad, "state" = 1, "ringZ" = RiseRingCurrentZ))

    if(EnableFreeze) {
        foreach(_:number, Ply:entity = ChargePlayers) {
            if(Ply:isValid()) { Ply:plyFreeze(1) }
        }
        foreach(_:number, Prop:entity = ChargeProps) {
            if(Prop:isValid()) { Prop:propManipulate(Prop:pos(), Prop:angles(), 1, 1, 0) }
        }
    }

    stoptimer("tp_charge")
    timer("tp_charge", ChargeTime, function() { doTpCharge() })
}

#==================================================
# EGP PRESS (needs runTeleport, so it lives here)
#==================================================

function void egpPress(Ply:entity) {
    if(!EnableEGPPicker) { return }
    if(!EGPBuilt) { return }
    if(!Ply:isValid()) { return }
    if(!EGP:entity():isValid()) { return }

    local Cursor = EGP:egpCursor(Ply)
    if(Cursor:x() < 0 | Cursor:y() < 0) { return }

    foreach(_:number, Btn:table = EGPButtons) {
        local Xeo = Btn["egpobject", egpobject]
        if(EGP:egpHasObject(Xeo) != 1) { continue }
        if(!Xeo:containsPoint(Cursor)) { continue }

        local PadEnt = Btn["pad", entity]
        entity():emitSound(HOLO_BUTTON_SOUND, HOLO_BUTTON_Volume, HOLO_BUTTON_Pitch, 1)
        stoptimer("egp_flash")
        timer("egp_flash", EGP_FlashDelay, function() { updateEGPColors() })

        if(!PadEnt:isValid() | !isWhitelisted(Ply) | Recharging == 1 | isOnCooldown(Ply)) {
            Xeo:setColor(VguiColorDenied)
            emitPadSound(SndDenied, SndDeniedVol, SndDeniedPitch)
            return
        }

        Xeo:setColor(VguiColorRecharge)
        SelectedPad = PadEnt
        runTeleport()
        return
    }
}

event chat(Player:entity, Message:string, _:number) {
    if(!Player:isValid()) { exit() }
    if(Player != owner()) { exit() }
    local LS = Message:explode(" ")
    if(LS:count() == 0) { exit() }
    if(LS[1,string]:sub(1,1) != ".") { exit() }
    hideChat(1)
    switch(LS[1,string]) {
        case ".list",
            local Count = Whitelist:count()
            if(Count == 0) {
                printColor(vec(0,200,255), "[Teleport Pad]: ", vec(255,255,255), "Whitelist is empty.")
                exit()
            }
            printColor(vec(0,200,255), "[Teleport Pad]: ", vec(255,255,255), "Whitelist (", vec(0,200,255), "" + Count, vec(255,255,255), "):")
            foreach(_:number, Allowed:entity = Whitelist) { 
                if(Allowed:isValid()) {
                    printColor(vec(0,200,255), "  - ", vec(255,255,255), Allowed:name())
                } else {
                    printColor(vec(0,200,255), "  - ", vec(255,80,80), "[offline/invalid entry]")
                }
            }
        break
        case ".clear",
            if(LS:count() == 1) {
                local OldCount = Whitelist:count()
                Whitelist = array()
                printColor(vec(0,200,255), "[Teleport Pad]: Whitelist cleared (", vec(255,255,255), "" + OldCount, vec(0,200,255), " removed).")
                exit()
            }
            local NameQ = LS[2,string]
            local I = 3
            while(I <= LS:count()) {
                NameQ = NameQ + " " + LS[I,string]
                I++
            }
            local Target = findPlayerByName(NameQ)
            if(Target:isPlayer()) {
                for(K = 1, Whitelist:count()) {
                    if(Whitelist[K,entity] == Target) {
                        Whitelist:remove(K)
                        printColor(vec(0,200,255), "[Teleport Pad]: Removed ", vec(255,255,255), Target:name())
                        exit()
                    }
                }
                printColor(vec(255,80,80), "[Teleport Pad]: ", vec(255,255,255), Target:name(), vec(255,80,80), " is not on the whitelist.")
                exit()
            }
            local NewWL = array()
            local Removed = 0
            local Query = NameQ:lower()
            foreach(_:number, Allowed:entity = Whitelist) {
                if(!Allowed:isValid()) { Removed++ continue }
                if(Allowed:name():lower():find(Query)) { Removed++ }
                else { NewWL:pushEntity(Allowed) }
            }
            Whitelist = NewWL
            if(Removed > 0) {
                local Ending = Removed == 1 ? "y" : "ies"
                printColor(vec(0,200,255), "[Teleport Pad]: Removed ", vec(255,255,255), "" + Removed, vec(0,200,255), " whitelist entr" + Ending + " matching '", vec(255,255,255), NameQ, vec(0,200,255), "'.")
            } else {
                printColor(vec(255,80,80), "[Teleport Pad]: No whitelist entries matched '", vec(255,255,255), NameQ, vec(255,80,80), "'.")
            }
        break
    }
}

event tick() {
    if(RingRGBMode & RGBTimerRunning & Ring1Idx > 0) {
        local IsMaster = 1
        foreach(IDStr:string, _:string = Partners) {
            if(IDStr:toNumber() < entity():id()) {
                IsMaster = 0
                break
            }
        }
        if(IsMaster) {
            local Hue = (curtime() * 200) % 360
            holoColor(Ring1Idx, hsv2rgb(Hue, 1, 1))
            RGBHue = Hue
            if(!timerExists("rgb_sync_send")) {
                timer("rgb_sync_send", RgbSyncSendInterval, function() { doRgbSyncSend() })
            }
        }
    }

    # Player presence backup (teli WIP.txt  keeps detection alive if clk timer stalls)
    if(Tele_Players) {
        foreach(_:number, Ply:entity = players()) {
            if(!Ply:isValid()) { continue }
            local SteamID = Ply:steamID()
            local Dist = Ply:pos():distance(entity():pos())

            if(!PlayersOnPad:exists(SteamID) & Dist <= UseTriggerDist) {
                PlayersOnPad[SteamID, number] = curtime()
                if(!TeleportedPlayers:exists(SteamID)) {
                    PendingGreets[SteamID, entity] = Ply
                    if(!timerExists("pad_greet")) { timer("pad_greet", PadGreetDelay, function() { doPadGreet() }) }
                    StepHoloPulse = 1
                }
            } elseif(PlayersOnPad:exists(SteamID) & Dist > LeavePadDist) {
                local ArrivalTime = PlayersOnPad[SteamID, number]
                if((curtime() - ArrivalTime) < 1) { continue }
                PlayersOnPad:remove(SteamID)
                TeleportedPlayers:remove(SteamID)
                PendingGreets:remove(SteamID)
                if(PickerPlayer == Ply) { closePickerVGUI() }
                PrevKeyUse[SteamID, number] = 0
                if(PlayersOnPad:count() == 0) {
                    StepHoloPulse = 0
                    if(holoEntity(35):isValid()) { holoEntity(35):setColor(StepHoloCol1) }
                }
            }
        }
    }

    foreach(_:number, Ply:entity = players()) {
        if(!Ply:isValid()) { continue }
        local SteamID = Ply:steamID()
        local CurKey  = Ply:keyUse()
        local PrevKey = PrevKeyUse[SteamID, number]
        PrevKeyUse[SteamID, number] = CurKey

        if(PrevKey & !CurKey) {
            setButtonColor(0)
            continue
        }
        if(!CurKey || PrevKey) { continue }

        if(EnableHoloButton && holoEntity(50):isValid()) {
            if(Ply:pos():distance(entity():pos()) <= Button_use_Dist) {
                if(isLookingAt(50, HoloButtonRadius, Ply)) {
                    entity():emitSound(HOLO_BUTTON_SOUND, HOLO_BUTTON_Volume, HOLO_BUTTON_Pitch, 1)
                    if(partnerCount() > 1) {
                        if(!isWhitelisted(Ply)) { emitPadSound(SndDenied, SndDeniedVol, SndDeniedPitch) break }
                        if(PickerOpen & PickerPlayer != Ply) { closePickerVGUI() }
                        openPickerVGUI(Ply)
                    } elseif(partnerCount() == 1) {
                        if(!isWhitelisted(Ply)) { emitPadSound(SndDenied, SndDeniedVol, SndDeniedPitch) break }
                        if(Recharging == 1 || isOnCooldown(Ply)) {
                            emitPadSound(SndDenied, SndDeniedVol, SndDeniedPitch)
                            setButtonColor(2)
                            stoptimer("button_reset")
                            timer("button_reset", ButtonDeniedResetTime, function() { doButtonReset() })
                            break
                        }
                        SelectedPad = firstPartner()
                        runTeleport()
                        setButtonColor(1)
                        stoptimer("button_reset")
                        timer("button_reset", ButtonResetFastDelay, function() { doButtonReset() })
                    } else {
                        entity():emitSound(SndCooldownDeny, SndCooldownDenyVol, SndCooldownDenyPitch)
                        setButtonColor(2)
                        stoptimer("button_reset")
                        timer("button_reset", ButtonDeniedResetTime, function() { doButtonReset() })
                    }
                    break
                }
            }
        }

        if(EnableUseTrigger && holoEntity(1):isValid()) {
            if(Ply:pos():distance(entity():pos()) <= UseButtonDist) {
                if(isLookingAt(1, RingUseRadius, Ply)) {
                    entity():emitSound(HOLO_BUTTON_SOUND, HOLO_BUTTON_Volume, HOLO_BUTTON_Pitch, 1)
                    if(partnerCount() > 1) {
                        if(!isWhitelisted(Ply)) { emitPadSound(SndDenied, SndDeniedVol, SndDeniedPitch) break }
                        if(PickerOpen & PickerPlayer != Ply) { closePickerVGUI() }
                        openPickerVGUI(Ply)
                    } elseif(partnerCount() == 1) {
                        if(!isWhitelisted(Ply)) { emitPadSound(SndDenied, SndDeniedVol, SndDeniedPitch) break }
                        if(Recharging == 1 || isOnCooldown(Ply)) {
                            emitPadSound(SndDenied, SndDeniedVol, SndDeniedPitch)
                            setButtonColor(2)
                            stoptimer("button_reset")
                            timer("button_reset", ButtonDeniedResetTime, function() { doButtonReset() })
                            break
                        }
                        SelectedPad = firstPartner()
                        runTeleport()
                    } else {
                        entity():emitSound(SndCooldownDeny, SndCooldownDenyVol, SndCooldownDenyPitch)
                        setButtonColor(2)
                        stoptimer("button_reset")
                        timer("button_reset", ButtonDeniedResetTime, function() { doButtonReset() })
                    }
                    break
                }
            }
        }
    }

    if(Spooling == 1) {
        C = clamp(C + Spool_up_delay, 0, Delay)
        SpinAdditive += C * C
        holoAng(30, holoEntity(1):toWorld(ang(0, SpinAdditive, 0)))
        soundPitch(2, C * Delay * SpoolPitchMult)
        if(EnableRiseRing & RiseRingCreated & holoEntity(60):isValid() & RiseRingAscendReady) {
            RiseRingCurrentZ = min(RiseRingCurrentZ + RiseRingAscendSpeed, RiseRingEndZ)
            holoPos(60, entity():toWorld(vec(0, 0, RiseRingCurrentZ)))
            holoAlpha(60, RiseRingAlpha)
        }
    } elseif(Spooling == 2) {
        C = clamp(C - Spool_down_delay, 0, Delay)
        SpinAdditive += C * C
        holoAng(30, holoEntity(1):toWorld(ang(0, SpinAdditive, 0)))
        soundPitch(2, C * Delay * SpoolPitchMult)
        if(EnableRiseRing & RiseRingCreated & holoEntity(60):isValid() & RiseRingDescendReady) {
            RiseRingCurrentZ = max(RiseRingCurrentZ - RiseRingDescendSpeed, RiseRingStartZ)
            holoPos(60, entity():toWorld(vec(0, 0, RiseRingCurrentZ)))
            holoAlpha(60, RiseRingAlpha)
        }
        if(RiseRingCurrentZ <= RiseRingStartZ & HeldPlayers:count() > 0) {
            HeldPlayers = array()
            ChargeActive = 0
        }
        if(C == 0 & RiseRingCurrentZ <= RiseRingStartZ) {
            Spooling = 0
            soundPitch(2, 0)
            RiseRingCurrentZ = RiseRingStartZ
            if(EnableRiseRing & RiseRingCreated & holoEntity(60):isValid()) {
                holoPos(60, entity():toWorld(vec(0, 0, RiseRingStartZ)))
                holoAlpha(60, RiseRingAlpha)
            }
            SelectedPad    = noentity()
            if(EnableFreeze) {
                foreach(_:number, Ply:entity = LastPortedPlayers) {
                    if(Ply:isValid()) { Ply:plyFreeze(0) }
                }
                foreach(_:number, Prop:entity = LastPortedProps) {
                    if(Prop:isValid()) { Prop:propManipulate(Prop:pos(), Prop:angles(), 0, 1, 0) }
                }
            }
        } elseif(C == 0) {
            soundPitch(2, 0)
        }
    }
}
